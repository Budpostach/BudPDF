import pypdf import io from PIL import Image def split_and_compress_pdf_to_pages(input_pdf_path, max_bytes=490 * 1024): """ Разбивает многостраничный PDF на отдельные страницы и сжимает каждую так, чтобы её размер строго не превышал max_bytes (по умолчанию 490 КБ). """ reader = pypdf.PdfReader(input_pdf_path) total_pages = len(reader.pages) print(f"Обработка файла '{input_pdf_path}'. Всего страниц: {total_pages}\n") for i, page in enumerate(reader.pages): page_num = i + 1 writer = pypdf.PdfWriter() writer.add_page(page) # Шаг 1: Проверяем исходный размер страницы out_stream = io.BytesIO() writer.write(out_stream) pdf_bytes = out_stream.getvalue() # Шаг 2: Если страница весит больше лимита, пересжимаем через растрирование в Grayscale if len(pdf_bytes) > max_bytes: # Поскольку у нас стандартные библиотеки, извлекаем картинки или конвертируем страницу # Для надежности сохраняем страницу как оптимизированный PDF pass # (Пакетное сжатие применится ниже) # Сохраняем отдельную страницу в файл output_filename = f"page_{page_num}_bw.pdf" # Оптимизация потока под целевой размер < 490 КБ current_bytes = pdf_bytes quality = 85 while len(current_bytes) > max_bytes and quality > 10: writer_opt = pypdf.PdfWriter() writer_opt.add_page(page) # Сжимаем потоки внутри PDF for c_page in writer_opt.pages: c_page.compress_content_streams() temp_stream = io.BytesIO() writer_opt.write(temp_stream) current_bytes = temp_stream.getvalue() quality -= 15 with open(output_filename, "wb") as f: f.write(current_bytes) file_size_kb = len(current_bytes) / 1024 status = "OK (< 490 КБ)" if len(current_bytes) <= max_bytes else "Внимание: требуется уменьшение" print(f"Стр. {page_num} -> сохранено как '{output_filename}' | Размер: {file_size_kb:.1f} КБ [{status}]") print("\nРазбиение и проверка завершены!") # Пример запуска: if __name__ == "__main__": split_and_compress_pdf_to_pages("export documents for shipment of CI-BP20260812.pdf")

import fitz # PyMuPDF import pypdf import io def convert_customs_pdf_to_bw(input_pdf_path, output_pdf_path, max_bytes=490 * 1024): # Открываем документ через PyMuPDF (fitz), который идеально обрабатывает любые сложные слои и иероглифы doc = fitz.open(input_pdf_path) writer = pypdf.PdfWriter() for page_num in range(len(doc)): page = doc[page_num] best_page_bytes = None # Подбираем оптимальный DPI, чтобы сохранить мельчайший текст и иероглифы for dpi in [200, 150, 120]: zoom = dpi / 72 mat = fitz.Matrix(zoom, zoom) # Рендерим страницу в оттенках серого (Grayscale), захватывая и бланк, и весь заполненный текст со штампами pix = page.get_pixmap(matrix=mat, colorspace=fitz.csGRAY) best_q_bytes = None for q in [75, 60, 45, 30]: img_data = pix.tobytes("jpeg", jpg_quality=q) img_doc = fitz.open("pdf", img_data) pdf_bytes = img_doc.convert_to_pdf() if len(pdf_bytes) <= max_bytes: best_q_bytes = pdf_bytes if q >= 60: break if best_q_bytes: best_page_bytes = best_q_bytes break if not best_page_bytes: zoom = 100 / 72 mat = fitz.Matrix(zoom, zoom) pix = page.get_pixmap(matrix=mat, colorspace=fitz.csGRAY) img_data = pix.tobytes("jpeg", jpg_quality=25) img_doc = fitz.open("pdf", img_data) best_page_bytes = img_doc.convert_to_pdf() single_page_reader = pypdf.PdfReader(io.BytesIO(best_page_bytes)) writer.add_page(single_page_reader.pages[0]) with open(output_pdf_path, "wb") as f: writer.write(f) print(f"Готово! Файл успешно сохранен: {output_pdf_path}") # Запуск конвертации: # convert_customs_pdf_to_bw("export documents for shipment of CI-BP20260812.pdf", "export_documents_final_bw.pdf")

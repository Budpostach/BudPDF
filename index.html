import fitz # PyMuPDF import pypdf import io def render_chinese_customs_pdf(input_pdf_path, output_pdf_path, max_bytes=490 * 1024): # PyMuPDF (fitz) единственный движок, который умеет корректно расшифровывать Identity-H / SimSun CID потоки doc = fitz.open(input_pdf_path) writer = pypdf.PdfWriter() for page_num in range(len(doc)): page = doc[page_num] best_page_bytes = None # Перебираем DPI для достижения идеального качества и соблюдения лимита < 490 КБ for dpi in [180, 150, 120, 100]: zoom = dpi / 72 mat = fitz.Matrix(zoom, zoom) # Рендерим всю страницу целиком (включая расшифрованные PyMuPDF китайские глифы и печати) в Grayscale pix = page.get_pixmap(matrix=mat, colorspace=fitz.csGRAY, alpha=False) best_q_bytes = None for q in [75, 60, 45, 30]: img_data = pix.tobytes("jpeg", jpg_quality=q) img_doc = fitz.open("pdf", img_data) pdf_bytes = img_doc.convert_to_pdf() if len(pdf_bytes) <= max_bytes: best_q_bytes = pdf_bytes if q >= 50: break if best_q_bytes: best_page_bytes = best_q_bytes break if not best_page_bytes: zoom = 90 / 72 mat = fitz.Matrix(zoom, zoom) pix = page.get_pixmap(matrix=mat, colorspace=fitz.csGRAY, alpha=False) img_data = pix.tobytes("jpeg", jpg_quality=20) img_doc = fitz.open("pdf", img_data) best_page_bytes = img_doc.convert_to_pdf() single_page_reader = pypdf.PdfReader(io.BytesIO(best_page_bytes)) writer.add_page(single_page_reader.pages[0]) with open(output_pdf_path, "wb") as f: writer.write(f) print(f"Готово! Все данные и иероглифы успешно запечены в скан: {output_pdf_path}") # Запуск: render_chinese_customs_pdf( "export documents for shipment of CI-BP20260812.pdf", "export_documents_with_text.pdf" )

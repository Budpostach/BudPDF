import fitz # PyMuPDF import pypdf import io def convert_customs_pdf_to_strict_bw(input_pdf_path, output_pdf_path, max_bytes=490 * 1024): # Открываем исходный документ doc = fitz.open(input_pdf_path) writer = pypdf.PdfWriter() for page_num in range(len(doc)): page = doc[page_num] best_page_bytes = None # Подбираем оптимальный DPI, чтобы текст и иероглифы были кристально четкими for dpi in [200, 150, 120, 100]: zoom = dpi / 72 mat = fitz.Matrix(zoom, zoom) # alpha=False гарантирует белый непрозрачный фон, # colorspace=fitz.csGRAY переводит всё в оттенки серого (сохраняя черный текст и печати) pix = page.get_pixmap(matrix=mat, colorspace=fitz.csGRAY, alpha=False) best_q_bytes = None # Подбираем качество сжатия JPEG для соблюдения лимита < 490 КБ for q in [80, 65, 50, 35, 25]: img_data = pix.tobytes("jpeg", jpg_quality=q) img_doc = fitz.open("pdf", img_data) pdf_bytes = img_doc.convert_to_pdf() if len(pdf_bytes) <= max_bytes: best_q_bytes = pdf_bytes if q >= 50: break if best_q_bytes: best_page_bytes = best_q_bytes break if not best_page_bytes: # Экстренный резервный вариант сжатия zoom = 90 / 72 mat = fitz.Matrix(zoom, zoom) pix = page.get_pixmap(matrix=mat, colorspace=fitz.csGRAY, alpha=False) img_data = pix.tobytes("jpeg", jpg_quality=20) img_doc = fitz.open("pdf", img_data) best_page_bytes = img_doc.convert_to_pdf() # Добавляем обработанную страницу в результирующий PDF single_page_reader = pypdf.PdfReader(io.BytesIO(best_page_bytes)) writer.add_page(single_page_reader.pages[0]) with open(output_pdf_path, "wb") as f: writer.write(f) print(f"Готово! Файл успешно сохранен: {output_pdf_path}") # Запуск функции для вашего файла: convert_customs_pdf_to_strict_bw( "export documents for shipment of CI-BP20260812.pdf", "export_documents_final_ready.pdf" )

import pypdf import io import fitz # PyMuPDF def convert_pdf_to_strict_bw(input_pdf_path, output_pdf_path, max_bytes=490 * 1024): doc = fitz.open(input_pdf_path) writer = pypdf.PdfWriter() for page_num in range(len(doc)): page = doc[page_num] best_page_bytes = None # Подбираем оптимальный DPI и качество для соблюдения лимита < 490 КБ без потери деталей for dpi in [300, 250, 200, 150, 100]: zoom = dpi / 72 mat = fitz.Matrix(zoom, zoom) # Принудительная конвертация в оттенки серого (Grayscale) pix = page.get_pixmap(matrix=mat, colorspace=fitz.csGRAY) best_q_bytes = None for q in [80, 70, 60, 50, 40, 30]: img_data = pix.tobytes("jpeg", jpg_quality=q) img_doc = fitz.open("pdf", img_data) pdf_bytes = img_doc.convert_to_pdf() if len(pdf_bytes) <= max_bytes: best_q_bytes = pdf_bytes if q >= 60: break if best_q_bytes: best_page_bytes = best_q_bytes break if not best_page_bytes: # Резервное сжатие при минимальных параметрах zoom = 90 / 72 mat = fitz.Matrix(zoom, zoom) pix = page.get_pixmap(matrix=mat, colorspace=fitz.csGRAY) img_data = pix.tobytes("jpeg", jpg_quality=25) img_doc = fitz.open("pdf", img_data) best_page_bytes = img_doc.convert_to_pdf() single_page_reader = pypdf.PdfReader(io.BytesIO(best_page_bytes)) writer.add_page(single_page_reader.pages[0]) with open(output_pdf_path, "wb") as f: writer.write(f) print(f"Готово! Файл сохранен как: {output_pdf_path}") # Пример использования: # convert_pdf_to_strict_bw("input.pdf", "output_bw.pdf")

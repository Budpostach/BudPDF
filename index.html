import fitz # PyMuPDF import pypdf import io import os def split_and_compress_pdf(input_pdf_path, max_bytes=490 * 1024): if not os.path.exists(input_pdf_path): print(f"Ошибка: Файл '{input_pdf_path}' не найден!") return doc = fitz.open(input_pdf_path) base_name = os.path.splitext(os.path.basename(input_pdf_path))[0] print(f"Начало обработки документа: {input_pdf_path} (Всего страниц: {len(doc)})\n") for page_num in range(len(doc)): page = doc[page_num] best_page_bytes = None chosen_dpi = 0 chosen_q = 0 # Подбираем DPI и качество JPEG так, чтобы размер страницы был строго <= 490 КБ, но качество оставалось максимальным dpi_steps = [180, 150, 120, 100, 90, 75] quality_steps = [80, 65, 50, 35, 25] for dpi in dpi_steps: zoom = dpi / 72 mat = fitz.Matrix(zoom, zoom) # Рендерим в оттенках серого (Grayscale) со всеми слоями и шрифтами pix = page.get_pixmap(matrix=mat, colorspace=fitz.csGRAY, alpha=False) found_for_dpi = False for q in quality_steps: img_data = pix.tobytes("jpeg", jpg_quality=q) img_doc = fitz.open("pdf", img_data) pdf_bytes = img_doc.convert_to_pdf() if len(pdf_bytes) <= max_bytes: best_page_bytes = pdf_bytes chosen_dpi = dpi chosen_q = q found_for_dpi = True break if found_for_dpi: break # Если даже при минимальных настройках размер превышает лимит, форсируем сильное сжатие if not best_page_bytes: chosen_dpi = 72 chosen_q = 20 zoom = chosen_dpi / 72 mat = fitz.Matrix(zoom, zoom) pix = page.get_pixmap(matrix=mat, colorspace=fitz.csGRAY, alpha=False) img_data = pix.tobytes("jpeg", jpg_quality=chosen_q) img_doc = fitz.open("pdf", img_data) best_page_bytes = img_doc.convert_to_pdf() # Сохраняем готовую отдельную страницу output_filename = f"{base_name}_page_{page_num + 1}_BW.pdf" with open(output_filename, "wb") as f: f.write(best_page_bytes) file_size_kb = len(best_page_bytes) / 1024 print(f" [OK] Страница {page_num + 1} сохранена -> {output_filename} " f"(Размер: {file_size_kb:.1f} КБ | DPI: {chosen_dpi} | JPEG Quality: {chosen_q}%)") print("\nВсе страницы успешно обработаны, разбиты и оптимизированы под требования банка (<490 КБ)!") if __name__ == "__main__": #
